// Idea: "Cuando llueve, el agua corre por el suelo, se mete al hoyo y lo desborda." Ground cut open, rain, water climbing inside the pit.
toma({mundo: 'oculto', fondo: '#dfe9ef'}, ({THREE, add, material, box, swarm, random, ramp, ease, palette, frame}) => {
	// The cut: three layers of earth, with a pit in the middle. Everything is built once, here.
	const layers = [palette.ground, '#a8643c', '#7d4a2c'];
	layers.forEach((color, i) => {
		const left = add(box(3, 1.2, 3, material(color)));
		left.position.set(-2.4, -1.2 * (i + 1), 0);
		const right = add(box(3, 1.2, 3, material(color)));
		right.position.set(2.4, -1.2 * (i + 1), 0);
	});
	const floor = add(box(7.8, 0.4, 3, material('#5e3720')));
	floor.position.set(0, -4, 0);
	const grass = add(box(7.8, 0.15, 3, material(palette.foliage)));
	grass.position.set(0, 0, 0);
	grass.scale.x = 1;

	// Water inside the pit: a box that grows from the bottom.
	const water = add(box(1.8, 1, 2.8, material(palette.water, {opacity: 0.85})));
	water.position.set(0, -3.6, 0);

	// Rain: many thin drops in one draw.
	const rand = random(7);
	const drops = add(swarm(new THREE.BoxGeometry(0.03, 0.35, 0.03), material('#9fd3ff', {glow: true}), 600));
	const seeds = Array.from({length: 600}, () => ({x: (rand() - 0.5) * 8, z: (rand() - 0.5) * 3, speed: 5 + rand() * 3, phase: rand()}));

	return (u, t) => {
		const rain = ramp(u, 0, 0.15);
		drops.each((o, i) => {
			const s = seeds[i];
			const fall = (s.phase + t * s.speed * 0.2) % 1;
			o.position.set(s.x, 6 - fall * 6.2, s.z);
			o.scale.setScalar(rain);
		});
		// The level rises once the rain has started, and spills over the edge at the end.
		const level = ease.inOut(ramp(u, 0.2, 0.95));
		water.scale.y = 0.2 + level * 3.6;
		water.scale.x = 1 + ramp(u, 0.85, 1) * 2.4;
		// Slow push in, framing the pit below the text band.
		const pose = frame([0, -1.8, 0], 4.2 - u * 0.6, {yaw: 0.25, pitch: 0.12, fov: 38});
		return {camera: pose};
	};
});
